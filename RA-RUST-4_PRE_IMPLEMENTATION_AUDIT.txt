//! The RA Software Platform entry binary.
//!
//! Per the frozen boot architecture the chain is:
//!
//! `main → Bootstrap::start → PlatformInterface::start → SySS::initialize →
//! Platform Kernel Initialization → Platform Ready`
//!
//! This binary carries no UI and no business logic: it starts the platform,
//! runs it, and shuts it down cleanly before the process exits. Failures
//! propagate to the process exit code.

use ra_bootstrap::Bootstrap;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let bootstrap = Bootstrap::start()?;
    bootstrap.shutdown()?;
    Ok(())
}

use crate::progress::{Progress, ProgressEvent};

/// The Bootstrap façade: starts and stops the RA platform.
///
/// A [`Bootstrap`] handle is obtained from [`Bootstrap::start`] and must be
/// torn down with [`Bootstrap::shutdown`] before the process exits.
///
/// The handle owns **no platform state** — no workspace, hubs, services or
/// lifecycle state. It only carries the opaque platform handle returned by
/// the [`Launcher`] so the platform can be shut down cleanly. It is not
/// `Copy`: a running platform has exactly one owner and one shutdown.
#[derive(Debug)]
pub struct Bootstrap {
    launcher: Launcher,
}

impl Bootstrap {
    /// Starts the RA platform.
    ///
    /// Runs the [`Startup`] sequence — minimal validation, progress events
    /// and [`Launcher::launch`] (which calls [`PlatformInterface::start`],
    /// the single public platform entry point) — and returns a [`Bootstrap`]
    /// handle for the running platform.
    ///
    /// TODO(Phase7B): the sequence is a contract placeholder; the real
    /// startup sequence is implemented in the SySS Hub Graph phase (Phase
    /// 7D.4 and later).
    pub fn start() -> Result<Bootstrap> {
        let launcher = Startup::run()?;
        Ok(Bootstrap { launcher })
    }

    /// Shuts the RA platform down cleanly.
    ///
    /// Emits shutdown progress events around the clean shutdown sequence
    /// (which delegates to [`PlatformInterface::shutdown`]). Phase 7F.5: the
    /// teardown ordering is real — `Bootstrap::shutdown →
    /// PlatformInterface::shutdown → SySS shutdown sequence → Kernel
    /// shutdown → Authorities shutdown → Platform stopped`. Bootstrap stays
    /// **orchestration only**: it delegates and owns no platform lifecycle
    /// logic.
    pub fn shutdown(self) -> Result<()> {
        Progress::emit(ProgressEvent::Stopping);
        let result = self.launcher.shutdown();
        if result.is_ok() {
            Progress::emit(ProgressEvent::Stopped);
        }
        result
    }
}

//! `ra-bootstrap`: the Bootstrap layer of the RA Software Platform.
//!
//! Bootstrap is the entry bridge between the operating system and the RA
//! platform. Per the frozen boot architecture the chain is:
//!
//! `Entry Point → Bootstrap::start → PlatformInterface::start →
//! SySS::initialize → Platform Kernel Initialization → Platform Ready`
//!
//! # Role
//!
//! * Bootstrap is **not** part of the Frontend.
//! * Bootstrap is **not** part of SySS.
//! * Bootstrap exists **only** to start the platform.
//! * Bootstrap owns **no platform state** (no workspace, hubs, services or
//!   lifecycle state) and performs **no business logic**.
//!
//! # Dependency rules
//!
//! Bootstrap depends **only** on `ra-syss` (and would use `ra-common` if the
//! shared vocabulary were required — it is not). It must **never** depend on
//! the IDE, Compiler, Runtime, Memory, Persistence, Database or Frontend.
//! Access to lower layers happens exclusively through the `ra-syss`
//! interface.
//!
//! # Surface
//!
//! * [`Bootstrap`] — the public façade: [`Bootstrap::start`] and
//!   [`Bootstrap::shutdown`].
//! * [`Startup`] — the startup sequence contract (validation → progress →
//!   launch).
//! * [`Launcher`] — launches the platform by calling
//!   [`PlatformInterface::start`] and carries the platform handle until
//!   shutdown.
//! * [`Validator`] — minimal pre-boot validation.
//! * [`Progress`] / [`ProgressEvent`] — boot progress events (contract only;
//!   no UI, terminal or animation).
//! * [`Shutdown`] — the clean-shutdown contract.
//!
//! Errors are reported through [`ra_syss::Result`]: at this stage every
//! bootstrap failure is a SySS-platform failure, so the two layers share one
//! error vocabulary.
//!
//! # Example
//!
//! ```
//! use ra_bootstrap::Bootstrap;
//!
//! let bootstrap = Bootstrap::start().unwrap();
//! bootstrap.shutdown().unwrap();
//! ```
//!
//! # Transitional status (Phase 7B)
//!
//! Bootstrap stays **orchestration only**: the startup sequence (validation
//! → progress → launch) is delegated to the platform, and the teardown
//! ordering is real since Phase 7F.5 — `Bootstrap::shutdown →
//! PlatformInterface::shutdown → SySS shutdown sequence → Kernel shutdown →
//! Authorities shutdown → Platform stopped` — implemented behind

//! The public Platform Interface (Phase 7C).
//!
//! [`PlatformInterface`] is the **single public platform API** of RA
//! Software. Per the frozen architecture it is the only surface exposed to
//! the Bootstrap, the Frontend, the future IDE, the future Database and the
//! future CLI: everything those consumers can ask of the platform — document
//! operations, execution control and the platform services — is expressed
//! here.
//!
//! # What is never exposed
//!
//! The interface deliberately never exposes implementation internals: the
//! hub graph, the message backbone, persistence internals, runtime internals
//! and compiler internals have no presence in this module or in its
//! signatures. Every operation is an architectural contract expressed in
//! [`crate::Result`]; the concrete platform behind the interface is
//! invisible to callers.
//!
//! # Current status
//!
//! The operations below began (Phase 7C) as documented placeholders that
//! performed no business logic; since then they have become real, phase by
//! phase. As of Phase 7D.3, [`PlatformInterface::start`] and
//! [`PlatformInterface::shutdown`]
//! drive the SySS platform skeleton behind the interface, and startup brings
//! the **Platform Kernel** online (authorities created, registered,
//! initialized and activated; Platform Ready). Phase 7E attaches the
//! **Platform Backbone** communication contracts behind the kernel; Phase
//! 7E.1 brings the **backbone runtime** online and wires the single proof
//! operation ([`PlatformInterface::probe`]). Phase 7F.1 wires the
//! **Authority → Service** proof ([`PlatformInterface::probe_service`]).
//! Phase 7F.2 wires the **service accessors** (workspace, project, package,
//! settings, history, log) through the real path — a minimal operation
//! travels the Platform Backbone to the owning authority, which forwards it
//! to its owned service. Phase 7F.3 turns the accessors into real service
//! operations: each accessor inspects the **real domain state** of its
//! service, which answers with the synchronous
//! `ServiceResult::Inspected` Response. Phase 7F.4 completes the
//! **Request → Response round-trip**: the accessors (and the proof operation
//! [`PlatformInterface::inspect_workspace`]) send a correlated Request over
//! the backbone, and the answer returns as an explicit `Response` message
//! delivered back over the backbone to the caller
//! (`Request → Backbone → Authority → Service → ServiceResult → Response →
//! Backbone → caller`). Phase 7F.5 adds the **platform lifecycle**:
//! [`PlatformInterface::suspend_platform`] pauses the platform (preserving
//! domain state; normal requests are rejected while suspended) and
//! [`PlatformInterface::resume_platform`] restores it;
//! [`PlatformInterface::shutdown`] performs the deterministic teardown
//! (`Bootstrap → PlatformInterface → SySS → Kernel → Authorities → Services
//! → Backbone → Platform State → Stopped`). Phase 7G audits and establishes
//! the **storage boundary foundation**: the Storage Service's domain state
//! represents the existing storage architecture (the `ra-syss` storage
//! stack — Syss storage space, Pss serialization, Recovery crash recovery —
//! over the storage vocabulary, built on `ra-persistence`), and the storage
//! Request → Response round-trip is proven internally. Phase 7G.1 adds the
//! **single public storage entry point** ([`PlatformInterface::storage`]):
//! the interface never exposes the Storage Service or the storage root, and
//! the real storage operations (Save / Load / Recover) ride the backbone
//! behind it. Phase 7H: the **document operations are real** —
//! [`PlatformInterface::open`], [`PlatformInterface::close`],
//! [`PlatformInterface::save`], [`PlatformInterface::save_as`],
//! [`PlatformInterface::import`] and [`PlatformInterface::export`] ride the
//! Request → Response path through the Platform Backbone to the Workspace
//! Authority, whose service manages the real Document active object over
//! the existing Syss/PSS storage stack. The workspace/project/package
//! active objects are managed through the same path behind the service
//! accessors. Since Phase 7H.1 those objects **persist** through the same
//! stack (into the existing `workspaces/`, `projects/` and `packages/`
//! categories): opening writes the state atomically, reopening restores it
//! (including the recorded parent association), closing and shutdown
//! preserve it, and corrupt or conflicting state is rejected — no
//! [`PlatformInterface`] signature changed. Phase 7H.2: **settings, history
//! and log are real persisted platform-state domains** behind the same
//! backbone path, and [`PlatformInterface::theme`] /
//! [`PlatformInterface::view`] are real settings-backed platform-state
//! operations (retrieval through the settings round-trip; updates through
//! the additive [`PlatformInterface::set_theme`] /
//! [`PlatformInterface::set_view`] / [`PlatformInterface::get_theme`] /
//! [`PlatformInterface::get_view`]). Theme/view are application state, not
//! UI rendering — the Frontend consumes them later. The execution-control
//! operations remain contracts (no-ops).
//!
//! Reviewed and stabilized in Phase 7C.1: no internal types leak into the
//! signatures (they use only `&Path` and [`crate::Result`]), no operations
//! were added or removed, and no premature abstractions (generics, service
//! traits, handles) were introduced.
//!
//! Per the frozen boot chain this interface is the **single public platform
//! entry point**: the Bootstrap calls [`PlatformInterface::start`], which
//! initializes the SySS platform skeleton behind the interface
//! ([`crate::SySS::initialize`]):
//!
//! `Entry Point → Bootstrap::start → PlatformInterface::start →
//! SySS::initialize → Platform Kernel Initialization → Platform Ready`
//!
//! Phase 7G.1: `PlatformInterface::start` also **establishes the platform
//! storage root** — `SySS::initialize` resolves, validates and creates the
//! storage root and the storage categories, runs startup crash recovery, and
//! fails the boot through [`crate::Error`] when the root cannot be
//! established. [`PlatformInterface::storage`] is the **single public
//! storage entry point**: it sends an inspection through the backbone to the

use std::path::{Path, PathBuf};

use crate::Result;

/// The public Platform Interface handle.
///
/// Obtained from [`PlatformInterface::start`] and shut down with
/// [`PlatformInterface::shutdown`]. It is the only public platform API;
/// consumers must not depend on anything else in `ra-syss` to operate the
/// platform.
///
/// The handle carries the SySS platform skeleton behind the interface — the
/// concrete platform is never exposed. It is not `Copy`: a running platform
/// has exactly one owner and one shutdown.
#[derive(Debug)]
pub struct PlatformInterface {
    platform: crate::SySS,
}

impl PlatformInterface {
    // ── Lifecycle ─────────────────────────────────────────────

    /// Starts the platform and returns a [`PlatformInterface`] handle.
    ///
    /// This is the single public platform entry point of the boot chain
    /// (`Bootstrap::start → PlatformInterface::start → SySS::initialize →
    /// Platform Kernel Initialization → Platform Ready`): it initializes the
    /// SySS platform skeleton behind the interface
    /// ([`crate::SySS::initialize`]), which brings the Platform Kernel online.
    ///
    /// TODO(Phase7C): startup drives the platform startup sequence, the
    /// Platform Kernel and the Platform Backbone contracts; platform services
    /// behind the interface arrive in Phase 7F and later.
    pub fn start() -> Result<PlatformInterface> {
        Self::start_with_storage_root(crate::syss::resolve_storage_root())
    }

    /// Starts the platform over the given storage root (Phase 7H, additive,
    /// crate-internal).
    ///
    /// The full startup sequence of [`PlatformInterface::start`] over `root`;
    /// the platform storage root is never exposed on the public surface — the
    /// tests use this constructor for isolation, mirroring
    /// [`crate::SySS::initialize_with_storage_root`].
    pub(crate) fn start_with_storage_root(root: PathBuf) -> Result<PlatformInterface> {
        let platform = crate::SySS::initialize_with_storage_root(root)?;
        Ok(PlatformInterface { platform })
    }

    /// Shuts the platform down, releasing everything it owns.
    ///
    /// Delegates to the SySS platform skeleton behind the interface, which
    /// performs the deterministic teardown ordering (Phase 7F.5): the
    /// platform state moves `Running | Suspended → ShuttingDown → Stopped`,
    /// and the kernel shuts down its whole authority set (`Active |
    /// Suspended → Shutdown`) — after shutdown no authority is Active, so
    /// the backbone rejects every normal platform message and no service
    /// can execute again. The handle is consumed: the platform is shut down
    /// exactly once and never reused.
    pub fn shutdown(self) -> Result<()> {
        self.platform.shutdown()
    }

    /// Suspends the platform (Phase 7F.5): pause, not shutdown.
    ///
    /// `PlatformInterface → SySS → Platform Kernel → Authorities → Services
    /// → Suspended`. After suspension the platform state is
    /// [`crate::Lifecycle::Suspended`], every authority is Suspended, and
    /// normal service requests travel the backbone to a rejection (no
    /// domain state is destroyed —
    /// [`PlatformInterface::resume_platform`] restores the same state).
    ///
    /// Named `suspend_platform` to distinguish the platform lifecycle from
    /// the frozen execution-control contracts ([`PlatformInterface::pause`] /
    /// [`PlatformInterface::resume`], which concern the active program, not
    /// the platform).
    pub fn suspend_platform(&mut self) -> Result<()> {
        self.platform.suspend()
    }

    /// Resumes the platform (Phase 7F.5).
    ///
    /// `PlatformInterface → SySS → Platform Kernel → Authorities → Active →
    /// Services Ready`. After resume the platform state is
    /// [`crate::Lifecycle::Running`], every authority is Active, services
    /// are ready again, and the domain state preserved during suspension is
    /// available as before. The counterpart of
    /// [`PlatformInterface::suspend_platform`].
    pub fn resume_platform(&mut self) -> Result<()> {
        self.platform.resume()
    }

    /// Probes the Platform Backbone communication path.
    ///
    /// Phase 7E.1: the **single minimal backbone-proof operation**. The call
    /// flows through the real communication path — a backbone message, target
    /// resolution and authority delivery — and returns the delivery result
    /// (`Ok(())` when the message was delivered and acknowledged).
    ///
    /// No business logic: this operation exists only to prove the backbone
    /// runtime. All other platform operations remain contracts.
    pub fn probe(&self) -> Result<()> {
        self.platform.probe_backbone()
    }

    /// Probes the Authority → Service path.
    ///
    /// Phase 7F.1: the proof that a platform request reaches a service
    /// through the backbone and its owning authority —
    /// `PlatformInterface → Backbone → LogAuthority → LogService →
    /// ServiceResult`. `Ok(())` means the Log proof service completed the
    /// request.
    ///
    /// No business logic: this operation exists only to prove the service
    /// path. All other platform operations remain contracts.
    pub fn probe_service(&self) -> Result<()> {
        // The proof request is a log command, which the Log proof service
        // completes; any other result means the proof path failed.
        match self.platform.probe_service()? {
            crate::hub::ServiceResult::Completed => Ok(()),
            _ => Err(crate::Error::Kernel(
                "log service did not complete the proof request".to_owned(),
            )),
        }
    }

    // ── Documents (Phase 7H) ──────────────────────────────────

    /// Opens a device file as the platform's active document.
    ///
    /// Phase 7H: the operation is real — a `Request` carrying the open
    /// document operation travels the Platform Backbone to the Workspace
    /// Authority, whose service brings the device file into the platform
    /// (the existing `Syss::import` *Device File → SySS* step), loads and
    /// deserializes it through the existing PSS stack, and opens the real
    /// [`Document`](crate::hub::Document) active object. `Ok(())` means the
    /// document is open (the answer is the correlated `Opened` response).
    ///
    /// The interface never touches the filesystem, PSS or active-object
    /// internals itself: everything stays behind the backbone.
    pub fn open(&self, path: &Path) -> Result<()> {
        let response = self.platform.request(
            crate::hub::HubId::WORKSPACE,
            crate::hub::ServiceOperation::OpenDocument {
                source: path.to_path_buf(),
            },
        )?;
        match response.payload() {
            crate::hub::ResponsePayload::Opened(_) => Ok(()),
            crate::hub::ResponsePayload::Rejected => Err(crate::Error::ActiveObject(
                "the document could not be opened".to_owned(),
            )),
            _ => Err(crate::Error::ActiveObject(
                "the document open did not answer with an opened document".to_owned(),
            )),
        }
    }

    /// Closes the platform's active document.
    ///
    /// Phase 7H: a `Request` carrying the close operation travels the

    pub fn save_as(&self, path: &Path) -> Result<()> {

    // ── Platform services ─────────────────────────────────────

    /// Requires a document operation to complete through the full Request →
    /// Response round-trip (Phase 7H).
    ///
    /// `PlatformInterface → Request → Backbone → Workspace Authority →
    /// Workspace Service → document object → ServiceResult → Response →
    /// Backbone → caller`. `Ok(())` means the document operation completed;
    /// a rejected response (an invalid target, a missing document, a
    /// persistence failure) is surfaced as an
    /// [`crate::Error::ActiveObject`] error — the operation is never
    /// reported as successful when the service rejected it.
    fn require_document_completed(
        &self,
        operation: crate::hub::ServiceOperation,
        message: &str,
    ) -> Result<()> {
        let response = self
            .platform
            .request(crate::hub::HubId::WORKSPACE, operation)?;
        match response.payload() {
            crate::hub::ResponsePayload::Completed => Ok(()),
            crate::hub::ResponsePayload::Rejected => {
                Err(crate::Error::ActiveObject(message.to_owned()))
            }
            _ => Err(crate::Error::ActiveObject(format!(
                "{message}: unexpected response"
            ))),
        }
    }

    /// Sends an inspection through the backbone to the service owned by the
    /// authority of `hub` and requires it to answer.
    ///
    /// The full round-trip of Phase 7F.4: `PlatformInterface → Request →
    /// Backbone → Authority → Service → ServiceResult → Response → Backbone
    /// → caller`. `Ok(())` means the owned service answered the inspection
    /// with a correlated `Response` carrying an inspected state summary. The
    /// summary itself is not surfaced at the interface level.
    fn require_inspected(&self, hub: crate::hub::HubId) -> Result<()> {
        match self
            .platform
            .request(hub, crate::hub::ServiceOperation::Inspect)?
            .payload()
        {
            crate::hub::ResponsePayload::Inspected(_) => Ok(()),
            _ => Err(crate::Error::Kernel(format!(
                "the service of hub {hub} did not answer the inspection"
            ))),
        }
    }

    /// Retrieves the settings value of `key` through the full Request →
    /// Response round-trip (Phase 7H.2): `PlatformInterface → Request →
    /// Backbone → Settings Authority → Settings Service → Syss/PSS →
    /// Response → caller`. The value is the persisted value or the
    /// deterministic default for a known setting; an unknown key or a
    /// corrupt settings store is surfaced as an
    /// [`crate::Error::ActiveObject`] error — never a fabricated value.
    fn settings_get(&self, key: &str) -> Result<String> {
        let response = self.platform.request(
            crate::hub::HubId::SETTINGS,
            crate::hub::ServiceOperation::Get(key.to_owned()),
        )?;
        match response.payload() {
            crate::hub::ResponsePayload::Inspected(value) => Ok(value.clone()),
            crate::hub::ResponsePayload::Rejected => Err(crate::Error::ActiveObject(format!(
                "the {key:?} setting is not available"
            ))),
            _ => Err(crate::Error::Kernel(format!(
                "the settings service did not answer the {key:?} retrieval"
            ))),
        }
    }

    /// Updates the settings value of `key` through the backbone (Phase 7H.2):
    /// the Settings Service validates and persists the value. A rejected
    /// update (an invalid or unsupported value) is surfaced as an
    /// [`crate::Error::ActiveObject`] error — never a fabricated success.
    fn settings_set(&self, key: &str, value: &str) -> Result<()> {
        match self.platform.invoke_service(
            crate::hub::HubId::SETTINGS,
            crate::hub::ServiceOperation::Set {
                key: key.to_owned(),
                value: value.to_owned(),
            },
        )? {
            crate::hub::ServiceResult::Completed => Ok(()),
            crate::hub::ServiceResult::Rejected => Err(crate::Error::ActiveObject(format!(
                "the {key:?} setting update was rejected"
            ))),
            _ => Err(crate::Error::Kernel(format!(
                "the settings service did not complete the {key:?} update"
            ))),
        }
    }

    /// Probes the full Request → Response round-trip (Phase 7F.4).
    ///
    /// The single proof operation of the round-trip contract: a `Request`
    /// carrying a fresh correlation identifier travels the Platform Backbone
    /// to the Workspace Authority, which forwards it to its owned Workspace
    /// Service; the service answers with its inspected state
    /// (`ServiceResult::Inspected`), which is converted into a correlated
    /// `Response` and delivered back through the backbone to the caller.
    /// `Ok(summary)` returns the inspected workspace state carried by the
    /// Response.
    ///
    /// `PlatformInterface → Request → Backbone → WorkspaceAuthority →
    /// WorkspaceService → ServiceResult → Response → Backbone → caller`
    ///
    /// No new domain behavior: the proof exists only to validate the
    /// communication contract.
    pub fn inspect_workspace(&self) -> Result<String> {
        let response = self.platform.request(
            crate::hub::HubId::WORKSPACE,
            crate::hub::ServiceOperation::Inspect,
        )?;
        match response.payload() {
            crate::hub::ResponsePayload::Inspected(summary) => Ok(summary.clone()),
            _ => Err(crate::Error::Kernel(
                "the workspace inspection did not answer with a summary".to_owned(),
            )),
        }
    }

    /// Requests access to the platform's workspace service.
    ///
    /// Phase 7F.3: an inspection of the real workspace state travels the
    /// Platform Backbone to the Workspace Authority, which forwards it to
    /// its owned Workspace Service; `Ok(())` means the service answered with
    /// its domain-state summary (the synchronous Response). No full
    /// workspace management yet.
    pub fn workspace(&self) -> Result<()> {
        self.require_inspected(crate::hub::HubId::WORKSPACE)

    /// Requests access to the platform's settings service.
    ///
    /// Phase 7F.3: an inspection of the real settings state travels the
    /// Platform Backbone to the Settings Authority, which forwards it to its

    pub fn settings(&self) -> Result<()> {
        self.require_inspected(crate::hub::HubId::SETTINGS)
    }

    /// Requests access to the platform's theme setting.
    ///
    /// Phase 7H.2: the theme is **settings-backed platform state** — the
    /// current value is retrieved through the Settings Service via the
    /// backbone (`Request → Backbone → SettingsAuthority → SettingsService →
    /// Syss/PSS → Response`). `Ok(())` means the theme is available (the
    /// deterministic default `light` when never set). No rendering: the
    /// Frontend consumes the platform state later, and RA Core is never
    /// involved.
    pub fn theme(&self) -> Result<()> {
        self.settings_get("theme").map(|_| ())
    }

    /// Returns the current platform theme through the settings round-trip
    /// (Phase 7H.2): the persisted value, or the deterministic default
    /// `light` when the theme was never set.
    pub fn get_theme(&self) -> Result<String> {
        self.settings_get("theme")
    }

    /// Sets the platform theme (Phase 7H.2): the value is validated and
    /// persisted by the Settings Service through the backbone. Supported
    /// values are `light` and `dark`; anything else is rejected
    /// deterministically.
    pub fn set_theme(&self, theme: &str) -> Result<()> {
        self.settings_set("theme", theme)
    }

    /// Requests access to the platform's view state.
    ///
    /// Phase 7H.2: the view is **settings-backed platform state** — the
    /// current view identity is retrieved through the Settings Service via
    /// the backbone. `Ok(())` means the view is available (the deterministic
    /// default `editor` when never set). This is application state, not a UI
    /// widget; no rendering and no GUI framework.
    pub fn view(&self) -> Result<()> {
        self.settings_get("view").map(|_| ())
    }

    /// Returns the current platform view identity through the settings
    /// round-trip (Phase 7H.2): the persisted value, or the deterministic

    pub fn get_view(&self) -> Result<String> {
        self.settings_get("view")
    }

    /// Sets the platform view identity (Phase 7H.2): the value is validated
    /// and persisted by the Settings Service through the backbone. Supported
    /// values are `editor`, `terminal` and `ide`; anything else is rejected

    pub fn set_view(&self, view: &str) -> Result<()> {
        self.settings_set("view", view)
    }

    /// Requests access to the platform's history service.
    ///
    /// Phase 7F.3: an inspection of the real history state travels the
    /// Platform Backbone to the History Authority, which forwards it to its
    /// owned History Service; `Ok(())` means the service answered with its
    /// domain-state summary (the synchronous Response). No complete history
    /// engine yet.
    pub fn history(&self) -> Result<()> {
        self.require_inspected(crate::hub::HubId::HISTORY)
    }

    /// Requests access to the platform's log service.
    ///
    /// Phase 7F.3: an inspection of the real log state travels the Platform
    /// Backbone to the Log Authority, which forwards it to its owned Log
    /// Service; `Ok(())` means the service answered with its domain-state
    /// summary (the synchronous Response). The Log Service is not yet a full
    /// logging subsystem.
    pub fn log(&self) -> Result<()> {
        self.require_inspected(crate::hub::HubId::LOG)
    }

    /// Requests access to the platform's storage service.
    ///
    /// Phase 7G.1: the **single public storage entry point**. An inspection
    /// of the real storage state travels the Platform Backbone to the
    /// Storage Authority, which forwards it to its owned Storage Service;
    /// `Ok(())` means the service answered with its domain-state summary
    /// (the synchronous Response). Real storage operations (Save / Load /

    /// proven by the platform tests.
    ///
    /// The interface never exposes the Storage Service or the storage root:
    /// every storage operation stays `PlatformInterface → Request → Backbone
    /// → StorageAuthority → StorageService`.
    pub fn storage(&self) -> Result<()> {
        self.require_inspected(crate::hub::HubId::STORAGE)
    }
}

=== SYSS PLATFORM ===
//! Platform foundation contracts (Phase 7A).
//!
//! The minimal, logic-free surface the Bootstrap builds on: the [`SySS`]
//! platform skeleton, the [`Lifecycle`] state representation and the
//! [`PlatformState`] holder. Per the frozen architecture, SySS is the
//! backbone of RA Software and owns the platform lifecycle and platform
//! state; nothing beyond state representation is implemented here, and
//! errors are reported through [`crate::Error`].
//!
//! TODO(Phase7D.3): the Platform Kernel is initialized on startup (created,
//! registered, initialized and activated hub authorities; Platform Ready) and
//! carries the Platform Backbone contracts (Phase 7E) and runtime (Phase
//! 7E.1 — [`SySS::probe_backbone`] proves the delivery path). Phase 7F.1:
//! [`SySS::probe_service`] proves the Authority → Service path. Phase 7F.2:
//! [`SySS::invoke_service`] drives the seven domain services through the
//! backbone with the minimal service-operation payload. Phase 7F.3:
//! [`SySS::invoke_service`] carries the real domain operations — an
//! inspection travels as a Request and is answered with the synchronous
//! [`ServiceResult::Inspected`] Response. Phase 7F.4:
//! [`SySS::request`] completes the **Request → Response round-trip** — a
//! Request carrying a correlation identifier travels the backbone to the
//! target authority, the owned service produces a [`ServiceResult`], the
//! explicit [`ResponsePayload::from_service_result`] boundary converts it
//! into a [`Response`] message, and the Response is delivered back over the
//! backbone to the requesting caller. Phase 7F.5: the **platform lifecycle
//! and safe shutdown** — the platform state moves through the validated
//! `Starting → Running ⇄ Suspended → ShuttingDown → Stopped` chain
//! ([`Lifecycle::can_transition`], [`PlatformState::transition`]);
//! [`SySS::suspend`]/[`SySS::resume`] pause and restore the platform
//! (preserving domain state) through the kernel's aggregate authority
//! control, and [`SySS::shutdown`] performs the deterministic teardown
//! ordering (stop new requests, shut down the kernel's authorities, stop
//! the platform state). Phase 7G: the **storage boundary audit & foundation**
//! — the Storage Service's domain state now represents the existing storage
//! architecture (the [`crate::Syss`] storage space, [`crate::Pss`]
//! serialization and [`crate::Recovery`] crash-recovery services over the
//! storage vocabulary), and the storage Request → Response round-trip,
//! correlation, error boundary and lifecycle gates are proven. Phase 7G.1:
//! **storage root + save/load/recovery integration** — [`SySS::initialize`]
//! resolves and establishes the platform storage root (validate, create the
//! root and the storage categories through [`Syss::ensure`], run startup
//! crash recovery through [`Recovery`], fail the boot on error) and owns the
//! [`Syss`] handle; the root is threaded down the storage boundary to the
//! Storage Authority, whose Storage Service performs the real `Save` /
//! `Load` / `Recover` operations through the existing stack, answered as
//! correlated Responses. Phase 7H: the **active objects** land on the
//! services — the workspace service owns the real [`Workspace`] and
//! [`Document`] objects and performs the real document operations (open,
//! close, save, save_as, import, export) through the existing Syss/PSS
//! stack, and the project/package services own their objects (with the
//! `Workspace → Project → Package` association) — all reached through
//! [`SySS::request`] and [`SySS::invoke_service`], which ride the same
//! Request → Response path. Phase 7H.1: the **workspace/project/package
//! active objects persist** through the existing Syss/PSS stack into the
//! existing `workspaces/`/`projects/`/`packages/` categories — the open
//! writes the state atomically, reopen restores it (including the recorded
//! parent association), close and shutdown preserve it, and corrupt or
//! conflicting state is rejected deterministically. Phase 7H.2: the
//! **platform-state domains** (Settings, History, Log) persist through the
//! same stack — settings as one file per key in `settings/`, history as
//! `history/history`, the smallest real structured platform log as
//! `logs/log` — lazy-loaded so corrupt state never fails the boot; theme
//! and view are settings-backed platform state with deterministic defaults
//! and validated values, surfaced through the interface. Still missing:
//! FSS (as a separate component), the device connector and the RA Core

use std::cell::RefCell;
use std::path::PathBuf;

use crate::Result;
use crate::hub::{

    HubId, Response, ResponsePayload, ServiceInvocation, ServiceOperation, ServiceResult,
};
use crate::{Recovery, Syss};

/// The SySS platform skeleton.
///
/// The minimal public surface of the RA platform backbone: the Bootstrap
/// creates it, initializes the platform through it, and shuts it down at
/// exit.
///
/// This is the **platform backbone**; it is distinct from [`crate::Syss`],
/// the System Storage Space (the artifact-storage subsystem). The
/// capitalization is deliberate: `SySS` = the platform, `Syss` = the storage
/// space.
///
/// Phase 7F.5: [`shutdown`](SySS::shutdown) performs the deterministic
/// teardown ordering (the platform state moves `Running | Suspended →
/// ShuttingDown → Stopped` and the kernel shuts down its authorities), and
/// [`suspend`](SySS::suspend)/[`resume`](SySS::resume) pause and restore the
/// platform while preserving domain state.
#[derive(Debug)]
pub struct SySS {
    // The platform state drives lifecycle reporting (Phase 7F.5): startup
    // moves it to Running, suspend/resume toggle Suspended, and shutdown
    // moves it through ShuttingDown to Stopped. The kernel is read by the
    // Phase 7E.1 backbone probe ([`SySS::probe_backbone`]) and driven by
    // the Phase 7F.5 aggregate lifecycle control.
    state: PlatformState,
    kernel: crate::hub::Kernel,
    // The platform storage root (Phase 7G.1): the platform owns the
    // resolved storage root — established at startup (validated, created,
    // recovered) and held as the [`Syss`] handle. The Storage Authority's
    // Storage Service performs the real storage operations over the same
    // root, threaded down the storage boundary at kernel construction.
    // Not yet read by the lib build: the tests verify the established root
    // (TODO(Phase7G.1)).
    #[allow(dead_code)]
    storage: Syss,
    // The Request → Response round-trip (Phase 7F.4) mints one correlation
    // identifier per Request from this sequence: deterministic, testable and
    // owned by the requesting endpoint (no global mutable state).
    request_seq: RefCell<CorrelationSequence>,
}

impl SySS {
    /// Initializes the RA platform and returns a handle to it.
    ///
    /// Runs the platform startup sequence:
    ///
    /// 1. **Platform State** — creates the platform state holder and moves
    ///    it to [`Lifecycle::Running`] once the platform is ready.
    /// 2. **Platform Kernel Initialization** — brings the Platform Kernel
    ///    online: creates, registers, initializes and activates the built-in
    ///    hub authorities, attaches the Platform Backbone contracts, then
    ///    verifies readiness.
    /// 3. **Platform Ready** — returns the running platform handle with its
    ///    authorities Active and its services ready.
    ///
    /// TODO(Phase7D.3): no backbone dispatch, no active objects, no services
    /// and no RA Core initialization — those are later phases.
    pub fn initialize() -> Result<SySS> {
        Self::initialize_with_storage_root(crate::syss::resolve_storage_root())
    }

    /// Initializes the platform over the given storage root (Phase 7G.1,
    /// additive).
    ///
    /// The full startup sequence over `root`:
    ///
    /// 1. **Platform Storage Root** — the platform owns the resolved storage
    ///    root. The root is validated and created through the existing
    ///    [`Syss::ensure`] (the root and every storage category directory),
    ///    then **startup crash recovery** runs through the existing
    ///    [`Recovery`] service (interrupted atomic writes are resolved). A
    ///    failure anywhere here fails the boot through the existing
    ///    [`crate::Error`] boundary — never silently ignored.
    /// 2. **Platform State** — creates the platform state holder and moves
    ///    it to [`Lifecycle::Running`] once the platform is ready.
    /// 3. **Platform Kernel Initialization** — brings the Platform Kernel
    ///    online over the same storage root (the root is threaded down the
    ///    storage boundary to the Storage Authority, so its Storage Service
    ///    performs real storage operations over the platform root): creates,
    ///    registers, initializes and activates the built-in hub authorities,
    ///    attaches the Platform Backbone contracts, then verifies readiness.
    /// 4. **Platform Ready** — returns the running platform handle with its
    ///    authorities Active and its services ready.
    pub(crate) fn initialize_with_storage_root(root: PathBuf) -> Result<SySS> {
        // 1. Platform Storage Root: validate + create the root and the
        //    storage categories (idempotent), then run startup crash
        //    recovery over the root.
        let storage = Syss::new(root.as_path());
        storage.ensure()?;
        let report = Recovery::new().recover(&root)?;
        if !report.failed.is_empty() {
            // Startup recovery could not resolve every interrupted write:
            // the boot fails through the existing error boundary instead of
            // silently continuing.
            return Err(crate::Error::Kernel(format!(
                "storage recovery failed on {} file(s)",
                report.failed.len()
            )));
        }
        // Platform State: created Starting, then moved to Running once the
        // platform is ready.
        let mut state = PlatformState::new();
        // Platform Kernel Initialization over the storage root.
        let kernel = crate::hub::KernelInitializer::new().initialize_with_storage(root)?;
        // Platform Ready: the platform is running (Phase 7F.5).
        state.transition(Lifecycle::Running)?;
        Ok(SySS {
            state,
            kernel,
            storage,
            request_seq: RefCell::new(CorrelationSequence::new()),
        })
    }

    /// Suspends the platform (Phase 7F.5): pause, not shutdown.
    ///
    /// Required path: `PlatformInterface → SySS → Platform Kernel →
    /// Authorities → Services → Suspended`. The platform state moves
    /// [`Lifecycle::Running`] → [`Lifecycle::Suspended`], then every
    /// built-in authority suspends `Active → Suspended` (the kernel owns the
    /// aggregate control). Suspension is deterministic: after it, the
    /// backbone's per-authority `Active` gate rejects every normal service
    /// request — **no domain state is destroyed**, and a later
    /// [`SySS::resume`] restores the same state.
    pub(crate) fn suspend(&mut self) -> Result<()> {
        // 1. The platform state reflects the suspension intent.
        self.state.transition(Lifecycle::Suspended)?;
        // 2. Every authority suspends together (all-or-nothing); the
        //    backbone gate then rejects normal traffic.
        self.kernel.suspend()?;
        Ok(())
    }

    /// Resumes the platform (Phase 7F.5).
    ///
    /// The platform state moves [`Lifecycle::Suspended`] →
    /// [`Lifecycle::Running`], then every built-in authority resumes
    /// `Suspended → Active`; the domain state preserved during suspension
    /// remains available and services are ready again.
    pub(crate) fn resume(&mut self) -> Result<()> {
        // 1. The platform state reflects the resume intent.
        self.state.transition(Lifecycle::Running)?;
        // 2. Every authority resumes together (all-or-nothing).
        self.kernel.resume()?;
        Ok(())
    }

    /// Shuts the platform down, releasing everything it owns (Phase 7F.5).
    ///
    /// Deterministic, synchronous teardown ordering:
    ///
    /// 1. **Stop accepting new normal requests** — the platform state moves
    ///    to [`Lifecycle::ShuttingDown`].
    /// 2. **Stop service execution and transition the authorities toward
    ///    Shutdown** — the kernel shuts down its whole authority set
    ///    (`Active | Suspended → Shutdown`, all-or-nothing); with no
    ///    `Active` authority left, the backbone gate rejects every normal
    ///    platform message (this also ends the backbone's participation).
    /// 3. **Shutdown the Platform State** — the state moves to
    ///    [`Lifecycle::Stopped`]; the handle is consumed, so the platform
    ///    handle is never reused and shutdown happens exactly once
    ///    (single-owner semantics — [`SySS`] is not `Copy`).
    ///
    /// The outer chain continues in the Bootstrap layer
    /// (`Bootstrap::shutdown → PlatformInterface::shutdown →
    /// SySS::shutdown`). No asynchronous teardown, no double shutdown: the
    /// handle is moved in and dropped.
    pub fn shutdown(mut self) -> Result<()> {
        // 1. Stop accepting new normal platform requests.
        self.state.transition(Lifecycle::ShuttingDown)?;
        // 2. Stop/suspend service execution: shut down the kernel's whole
        //    authority set (authorities Shutdown, services can never execute
        //    again, the backbone rejects all normal traffic).
        self.kernel.shutdown()?;
        // 3. Shutdown the platform state.
        self.state.transition(Lifecycle::Stopped)?;
        Ok(())
    }

    /// Probes the Platform Backbone communication path.
    ///
    /// Phase 7E.1: the minimal real communication path. A platform
    /// notification is sent from the Device hub (the host-facing hub) through
    /// the Platform Backbone to the Log hub — envelope validation, target
    /// resolution and authority delivery. `Ok(())` means the message was
    /// delivered and acknowledged. No business logic.
    pub(crate) fn probe_backbone(&self) -> Result<()> {
        let message = crate::hub::BackboneMessage::new(
            crate::hub::BackboneMessageKind::Notification,
            crate::hub::HubId::DEVICE,
            Some(crate::hub::HubId::LOG),
        );
        self.kernel.backbone().dispatch(message)?;
        Ok(())
    }

    /// Probes the Authority → Service path.
    ///
    /// Phase 7F.1: the proof that a platform request reaches a service
    /// through the backbone and its owning authority. A log request is sent
    /// from the Device hub through the Platform Backbone to the Log hub
    /// (delivery), then forwarded by the Log Authority to its owned Log
    /// proof service, which returns its [`ServiceResult`]. No business
    /// logic.
    pub(crate) fn probe_service(&self) -> Result<ServiceResult> {
        // The Log proof service completes a log request (Phase 7F.1).
        self.invoke_service(HubId::LOG, ServiceOperation::Accept)
    }

    /// Invokes the service owned by the authority of `hub` with the given
    /// operation, through the real communication path.
    ///
    /// Phases 7F.2–7F.3: a platform request carrying `operation` is sent
    /// from the Device hub (the host-facing hub, the established platform
    /// entry) through the Platform Backbone to `hub` — envelope validation,
    /// target resolution and delivery — then forwarded by the target
    /// authority to its owned service, which performs the real domain
    /// operation and returns its [`ServiceResult`]. An inspection travels as
    /// a `Request` and is answered with the synchronous
    /// [`ServiceResult::Inspected`] Response; every other real operation
    /// travels as a `Command` (a directive).
    ///
    /// Exception: when `hub` is the Device hub itself the request is sourced
    /// from the Workspace hub instead — the routing contract forbids
    /// self-routes, so the Device service cannot be invoked from the Device
    /// hub.
    pub(crate) fn invoke_service(
        &self,
        hub: HubId,
        operation: ServiceOperation,
    ) -> Result<ServiceResult> {
        // The platform entry is the host-facing Device hub, except when the
        // target is the Device hub itself — the routing contract forbids
        // self-routes, so the Device probe is sourced from the Workspace hub
        // instead.
        let source = if hub == HubId::DEVICE {
            HubId::WORKSPACE
        } else {
            HubId::DEVICE
        };
        // An inspection travels as a Request (it expects an answer — the
        // synchronous Response); every other real operation travels as a
        // Command (a directive).
        let kind = if matches!(operation, ServiceOperation::Inspect) {
            BackboneMessageKind::Request
        } else {
            BackboneMessageKind::Command
        };
        let message = BackboneMessage::new(kind, source, Some(hub))
            .with_payload(BackbonePayload::Operation(operation));
        // 1. The Platform Backbone delivers the request (transport and
        //    target resolution; the target authority acknowledges receipt).
        //    The message is cloned into dispatch: it is read again below to
        //    build the invocation (the envelope is not `Copy`).
        self.kernel.backbone().dispatch(message.clone())?;
        // 2. The target authority forwards the request to its owned service.
        //    The built-in authority set always contains every hub on a ready
        //    kernel; the error below is defensive and deterministic.
        let authority = self
            .kernel
            .authorities()
            .iter()
            .find(|authority| authority.id() == hub)
            .ok_or(HubError::UnregisteredTarget(hub))?;
        let invocation = ServiceInvocation::new(&message);
        Ok(authority.invoke(&invocation)?)
    }

    /// Runs the full Request → Response round-trip (Phase 7F.4):
    ///
    /// `caller → Request → Backbone → Target Authority → Owned Service →
    /// ServiceResult → Response → Backbone → caller`
    ///
    /// A Request carrying `operation` and a fresh [`CorrelationId`] is
    /// delivered to `hub`'s authority through the Platform Backbone
    /// (`dispatch` — envelope validation, target resolution and delivery);
    /// the authority forwards it to its owned service; the resulting
    /// [`ServiceResult`] is converted into a [`Response`] through the
    /// explicit [`ResponsePayload::from_service_result`] boundary; the
    /// Response is verified against the Request's correlation and delivered
    /// back to the caller through the backbone (`reply` — the existing
    /// routing and validation); and the Response is returned to the caller.
    ///
    /// Synchronous and deterministic — no queues, no asynchronous runtime,
    /// no networking, no message persistence.
    ///
    /// The envelope is always a `Request` (a question expecting an answer).
    /// Mutating operations (`Open`, `Set`, `Record`, ...) are Command-kind
    /// directives and travel through [`SySS::invoke_service`] instead; a
    /// mutating operation offered to the round-trip is answered with a
    /// `Rejected` response by the service's kind contract.
    ///
    /// Exception (Phase 7G.1): the **storage** domain's real operations —
    /// `Save`, `Load`, `Recover` — are request/response operations and
    /// travel through this round-trip as Requests, answered with the typed
    /// storage result (`Completed`, `Loaded`, `Recovered`) or `Rejected`.
    ///
    /// Exception: when `hub` is the Device hub itself the request is sourced
    /// from the Workspace hub instead — the routing contract forbids
    /// self-routes, so the Device service cannot be asked from the Device
    /// hub.
    pub(crate) fn request(&self, hub: HubId, operation: ServiceOperation) -> Result<Response> {
        // The platform entry is the host-facing Device hub, except when the
        // target is the Device hub itself — the routing contract forbids
        // self-routes, so the Device round-trip is sourced from the Workspace
        // hub instead.
        let source = if hub == HubId::DEVICE {
            HubId::WORKSPACE
        } else {
            HubId::DEVICE
        };
        let correlation = self.request_seq.borrow_mut().next();
        let request = BackboneMessage::new(BackboneMessageKind::Request, source, Some(hub))
            .with_correlation(correlation)
            .with_payload(BackbonePayload::Operation(operation));
        // 1. Deliver the Request through the backbone (transport and target
        //    resolution; the target authority acknowledges receipt). The
        //    message is cloned into dispatch: it is read again below.
        self.kernel.backbone().dispatch(request.clone())?;
        // 2. The target authority forwards the request to its owned service.
        //    The built-in authority set always contains every hub on a ready
        //    kernel; the error below is defensive and deterministic.
        let authority = self
            .kernel
            .authorities()
            .iter()
            .find(|authority| authority.id() == hub)
            .ok_or(HubError::UnregisteredTarget(hub))?;
        let service_result = authority.invoke(&ServiceInvocation::new(&request))?;
        // 3. Convert the ServiceResult into a Response addressed back to the
        //    requesting caller, echoing the Request's correlation (the
        //    explicit ServiceResult → Response mapping boundary).
        let response = Response::new(
            hub,
            source,
            Some(correlation),
            ResponsePayload::from_service_result(service_result),
        );
        // 4. Verify the correlation: the round-trip built both sides, so a
        //    mismatch would mean a foreign response — rejected defensively
        //    and deterministically.
        response.verify_answers(correlation)?;
        // 5. Deliver the Response back to the caller through the backbone
        //    (the existing `reply` routing and validation).
        self.kernel.backbone().reply(response.message().clone())?;
        Ok(response)
    }
}

/// The platform lifecycle phases.
///
/// The platform lifecycle of the whole RA platform, driven by [`SySS`]
/// (Phase 7F.5): `Starting → Running ⇄ Suspended → ShuttingDown → Stopped`.
/// Distinct from the per-hub [`HubLifecycle`](crate::hub::HubLifecycle): this
/// describes the life of the platform as a whole, not of a single hub.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Lifecycle {
    /// The platform is being brought up.
    Starting,
    /// The platform is operational (authorities Active, services ready).
    Running,
    /// The platform is **suspended**: paused, not shut down — domain state
    /// is preserved and normal requests are rejected (Phase 7F.5).
    Suspended,
    /// The platform is being torn down; no new requests are accepted.
    ShuttingDown,
    /// The platform has fully stopped; the handle has been consumed.
    Stopped,
}

impl Lifecycle {
    /// Returns whether `self` may transition to `to` under the validated
    /// platform lifecycle contract (Phase 7F.5).
    ///
    /// The single transition table of the platform lifecycle:
    ///
    /// ```text
    /// Starting     → Running
    /// Running      → Suspended
    /// Suspended    → Running
    /// Running      → ShuttingDown
    /// Suspended    → ShuttingDown
    /// ShuttingDown → Stopped
    /// ```
    ///
    /// [`Lifecycle::Stopped`] is terminal; every other pair is rejected.
    pub(crate) fn can_transition(self, to: Lifecycle) -> bool {
        matches!(
            (self, to),
            (Lifecycle::Starting, Lifecycle::Running)
                | (Lifecycle::Running, Lifecycle::Suspended)
                | (Lifecycle::Suspended, Lifecycle::Running)
                | (Lifecycle::Running, Lifecycle::ShuttingDown)
                | (Lifecycle::Suspended, Lifecycle::ShuttingDown)
                | (Lifecycle::ShuttingDown, Lifecycle::Stopped)
        )
    }
}

/// The platform state contract.
///
/// Holds the validated [`Lifecycle`] phase of the platform (Phase 7F.5);
/// workspace, hubs, services and active objects are added in later phases.
///
/// The state is created [`Lifecycle::Starting`], moved to
/// [`Lifecycle::Running`] by [`SySS::initialize`], toggled between
/// [`Lifecycle::Suspended`] and [`Lifecycle::Running`] by
/// [`SySS::suspend`]/[`SySS::resume`], and moved through
/// [`Lifecycle::ShuttingDown`] to [`Lifecycle::Stopped`] by
/// [`SySS::shutdown`] — always through the validated [`transition`]
/// (PlatformState::transition) boundary.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PlatformState {
    lifecycle: Lifecycle,
}

impl PlatformState {
    /// Returns the initial platform state, freshly created and
    /// [`Lifecycle::Starting`].
    pub fn new() -> PlatformState {
        PlatformState {
            lifecycle: Lifecycle::Starting,
        }
    }

    /// Returns the current lifecycle phase.
    pub fn lifecycle(&self) -> Lifecycle {
        self.lifecycle
    }

    /// The validated platform lifecycle transition (Phase 7F.5).
    ///
    /// Applies the single transition table ([`Lifecycle::can_transition`]);
    /// an invalid transition is rejected with a deterministic
    /// [`crate::Error::Kernel`] error and the state is left unchanged. Used
    /// by [`SySS`] as the platform drives its lifecycle; the state is never
    /// mutated directly.
    pub(crate) fn transition(&mut self, to: Lifecycle) -> Result<()> {
        let from = self.lifecycle;
        if !from.can_transition(to) {
            return Err(crate::Error::Kernel(format!(
                "platform cannot transition from {from:?} to {to:?}"
            )));
        }
        self.lifecycle = to;
        Ok(())
    }
}

impl Default for PlatformState {
    fn default() -> PlatformState {
        PlatformState::new()
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn initialize_returns_a_platform_handle() {
        assert!(SySS::initialize().is_ok());
    }

    #[test]
    fn initialize_brings_the_platform_kernel_online() {
        let platform = SySS::initialize().unwrap();
        // The Platform Kernel must be Platform Ready after initialization,
        // and the platform state reflects the running platform.
        assert!(platform.kernel.is_ready());
        assert_eq!(platform.state.lifecycle(), Lifecycle::Running);
    }

    #[test]
    fn shutdown_completes() {
        let platform = SySS::initialize().unwrap();
        assert!(platform.shutdown().is_ok());
    }

    #[test]
    fn probe_service_reaches_the_log_proof_service() {
        let platform = SySS::initialize().unwrap();
        assert_eq!(
            platform.probe_service().unwrap(),
            crate::hub::ServiceResult::Completed
        );
    }

    #[test]
    fn invoke_service_completes_for_every_domain_hub() {
        let platform = SySS::initialize().unwrap();
        for hub in [
            crate::hub::HubId::WORKSPACE,
            crate::hub::HubId::PROJECT,
            crate::hub::HubId::PACKAGE,
            crate::hub::HubId::SETTINGS,
            crate::hub::HubId::HISTORY,
            crate::hub::HubId::LOG,
            crate::hub::HubId::STORAGE,
            crate::hub::HubId::DEVICE,
        ] {
            assert_eq!(
                platform
                    .invoke_service(hub, crate::hub::ServiceOperation::Accept)
                    .unwrap(),
                crate::hub::ServiceResult::Completed,
                "the service of {hub} must complete its minimal operation"
            );
        }
    }

    #[test]
    fn real_workspace_operations_flow_through_the_backbone() {
        let (_dir, platform) = temp_platform();
        // Phase 7F.3: a real domain operation travels the real path —
        // Backbone → Workspace Authority → Workspace Service → domain state
        // → ServiceResult.
        assert_eq!(
            platform
                .invoke_service(
                    crate::hub::HubId::WORKSPACE,
                    crate::hub::ServiceOperation::Open("ra-main".to_owned())
                )
                .unwrap(),
            crate::hub::ServiceResult::Completed
        );
        // The inspection is answered with the real domain state — the
        // synchronous Request → Response path.
        assert_eq!(
            platform
                .invoke_service(
                    crate::hub::HubId::WORKSPACE,
                    crate::hub::ServiceOperation::Inspect
                )
                .unwrap(),
            crate::hub::ServiceResult::Inspected("workspace: ra-main".to_owned())
        );
    }

    #[test]
    fn request_returns_a_correlated_response_for_workspace_inspect() {
        let platform = SySS::initialize().unwrap();
        // The full Phase 7F.4 round-trip: Request → Backbone → Workspace
        // Authority → Workspace Service → ServiceResult → Response →
        // Backbone → caller.
        let response = platform
            .request(
                crate::hub::HubId::WORKSPACE,
                crate::hub::ServiceOperation::Inspect,
            )
            .unwrap();
        // The Response answers the Request: correlation set, targets
        // reversed, and the inspected state carried back to the caller.
        assert!(response.correlation().is_some());
        assert_eq!(response.kind(), crate::hub::BackboneMessageKind::Response);
        assert_eq!(response.source(), crate::hub::HubId::WORKSPACE);
        assert_eq!(response.target(), Some(crate::hub::HubId::DEVICE));
        assert!(matches!(
            response.payload(),
            crate::hub::ResponsePayload::Inspected(summary)
                if summary == "workspace: none"
        ));
    }

    #[test]
    fn request_responses_reflect_real_domain_state() {
        let (_dir, platform) = temp_platform();
        // Open the workspace through the direct path, then observe the real
        // state through the full round-trip.
        platform
            .invoke_service(
                crate::hub::HubId::WORKSPACE,
                crate::hub::ServiceOperation::Open("ra-main".to_owned()),
            )
            .unwrap();
        let response = platform
            .request(
                crate::hub::HubId::WORKSPACE,
                crate::hub::ServiceOperation::Inspect,
            )
            .unwrap();
        assert!(matches!(
            response.payload(),
            crate::hub::ResponsePayload::Inspected(summary)
                if summary == "workspace: ra-main"
        ));
    }

    #[test]
    fn correlation_ids_do_not_leak_between_requests() {
        let platform = SySS::initialize().unwrap();
        let first = platform
            .request(
                crate::hub::HubId::WORKSPACE,
                crate::hub::ServiceOperation::Inspect,
            )
            .unwrap();
        let second = platform
            .request(
                crate::hub::HubId::WORKSPACE,
                crate::hub::ServiceOperation::Inspect,
            )
            .unwrap();
        // Every Request gets its own correlation identifier; each Response
        // answers exactly its own Request.
        assert_ne!(first.correlation(), second.correlation());
        first.verify_answers(first.correlation().unwrap()).unwrap();
        second
            .verify_answers(second.correlation().unwrap())
            .unwrap();
    }

    #[test]
    fn a_rejected_operation_round_trips_as_a_rejected_response() {

        assert!(state.transition(Lifecycle::Suspended).is_err());
        // Starting → ShuttingDown is invalid.
        assert!(state.transition(Lifecycle::ShuttingDown).is_err());
        state.transition(Lifecycle::Running).unwrap();
        // Running → Running is invalid.
        assert!(state.transition(Lifecycle::Running).is_err());
        state.transition(Lifecycle::Suspended).unwrap();
        // Suspended → ShuttingDown is valid (shutdown from suspension);
        // Suspended → Suspended is not.
        assert!(state.transition(Lifecycle::Suspended).is_err());
        state.transition(Lifecycle::ShuttingDown).unwrap();
        // ShuttingDown → Running is invalid.
        assert!(state.transition(Lifecycle::Running).is_err());
        state.transition(Lifecycle::Stopped).unwrap();
        // Stopped is terminal.
        assert!(state.transition(Lifecycle::Running).is_err());
        assert!(state.transition(Lifecycle::Stopped).is_err());

=== SYSS LIBRARY ===
//! `ra-syss`: the SySS layer of the RA Software Platform.
//!
//! Per the frozen architecture, SySS is the backbone and backend of RA
//! Software: all communication between the Frontend and RA Core passes through
//! it, and it owns the platform lifecycle, workspace, active objects, platform
//! state and platform services.
//!
//! This crate hosts the storage-oriented SySS services migrated from
//! `ra-persistence` and the minimal platform-foundation contracts:
//!
//! * **SySS — System Storage Space** ([`Syss`]): the persistent storage
//!   subsystem. It stores workspaces, projects, packages, loaded files,
//!   settings, logs, history and serialized artifacts, and performs all
//!   writes atomically.
//! * **PSS — Program Serialization Service** ([`Pss`]): the service that
//!   serializes and deserializes RA program artifacts by dispatching to the
//!   codec layer owned by `ra-persistence`.
//! * **Recovery** ([`Recovery`]): crash recovery for interrupted SySS writes.
//! * **Platform foundation** ([`SySS`], [`Lifecycle`], [`PlatformState`]):
//!   the minimal, logic-free contract surface the Bootstrap builds on — the
//!   platform skeleton, the lifecycle phases and the platform state holder.
//! * **Storage categories** ([`StorageCategory`]): the storage vocabulary —
//!   workspaces, projects, packages, loaded files, settings, logs, history
//!   and artifacts — owned by SySS since Phase 7C.1.
//! * **Platform Interface** ([`PlatformInterface`]): the single public
//!   platform API — the only surface exposed to the Bootstrap, the Frontend,
//!   the future IDE, Database and CLI. Contracts only; hub graph, message
//!   backbone and layer internals are never exposed.
//! * **Hub Graph contracts** (Phase 7D.1), **hub authorities** (Phase 7D.2),
//!   the **Platform Kernel** (Phase 7D.3) and the **Platform Backbone**
//!   (Phases 7E–7E.1): the internal `hub` module — hub identifiers, hub domains,
//!   the hub lifecycle vocabulary, the hub registry, the hub graph knowledge,
//!   the routing contracts, the concrete per-domain hub authorities, the
//!   kernel initializer that brings them online (create, register,
//!   initialize, activate, verify readiness) and the communication contracts
//!   the backbone consumes. Architectural scaffolding only, kept strictly
//!   behind the platform interface.
//!
//! # Dependency direction
//!
//! `ra-syss` depends on `ra-persistence` (for the artifact model, the codec
//! layer and the error/format vocabulary). `ra-persistence` does **not**
//! depend back on `ra-syss`, so the crate graph stays acyclic.
//!
//! # Transitional status (Phases 7A–7D)
//!
//! TODO(Phase7A): the platform types ([`SySS`], [`Lifecycle`],
//! [`PlatformState`]) are **contracts only** — no platform logic exists.
//!
//! TODO(Phase7C): the [`PlatformInterface`] is the same kind of contract —
//! every operation is a documented no-op. Since Phase 7C.1 the storage
//! services ([`Syss`], [`Pss`], [`Recovery`]) report errors through
//! [`Error`], and the storage vocabulary and storage-oriented error variants
//! are owned here.
//!
//! TODO(Phase7D): the **SySS Hub Graph** phase realizes the contracts: the
//! message backbone / hub graph, active object management, FSS and the
//! device connector. Phase 7D.1 landed the hub graph **contracts**, Phase
//! 7D.2 the concrete **hub authorities**, Phase 7D.3 the **Platform Kernel**
//! initialization and Phase 7D.3.1 the **stabilization** pass (naming,
//! visibility, ownership and documentation freeze) — all in the internal
//! `hub` module (see above). Phase 7E attaches the **Platform Backbone**
//! communication contracts (vocabulary, envelope and interface) as the first
//! consumer of the kernel's registry, graph, routing and authorities. Phase
//! 7E.1 turns the contracts into the **minimum real backbone runtime** —
//! deterministic synchronous delivery, target resolution and delivery
//! receipts, proven through [`PlatformInterface::probe`]. Phase 7F.1 lands
//! the **service architecture** — the common Service abstraction and the
//! single Log proof service owned by the Log Authority, proven through
//! [`PlatformInterface::probe_service`] (`Backbone → Authority → Service →
//! ServiceResult`). Phase 7F.2 lands the **Platform Services** — the seven
//! domain services (workspace, project, package, settings, history,
//! storage, device) bound to their authorities, performing minimal
//! deterministic domain operations and carrying the minimal
//! service-operation payload; the service accessors of
//! [`PlatformInterface`] (workspace, project, package, settings, history,
//! log) are wired through the backbone. Phase 7F.3 turns the services into
//! the first **real domain-state services**: each service owns private
//! domain state, performs real domain operations (`Open`/`Close`, `Set`,
//! `Record`/`Clear`, `Probe`) and answers a `Request` carrying `Inspect`
//! with the synchronous `ServiceResult::Inspected` Response; the service
//! accessors inspect that real state. Phase 7F.4 completes the **Request →
//! Response round-trip**: a Request carries an opaque correlation
//! identifier, the service result is converted into an explicit `Response`
//! message through the `ResponsePayload::from_service_result` boundary and
//! delivered back over the backbone to the caller. Phase 7F.5 completes the
//! **platform lifecycle & safe shutdown**: the platform state moves through
//! the validated `Starting → Running ⇄ Suspended → ShuttingDown → Stopped`
//! chain ([`Lifecycle`], [`PlatformState`]); [`SySS::suspend`]/
//! [`SySS::resume`] pause and restore the platform while preserving domain
//! state, and [`SySS::shutdown`] performs the deterministic teardown
//! ordering (stop new requests → shut down the kernel's authorities →
//! services stop → backbone participation ends → platform state stopped),
//! keeping the single-owner shutdown semantics through the Bootstrap chain.
//! Phase 7G: the **storage boundary audit & foundation** — the Storage
//! Service's domain state represents the existing storage architecture (the
//! [`Syss`] storage space, [`Pss`] serialization and [`Recovery`] crash
//! recovery over the [`StorageCategory`] vocabulary, built on the
//! `ra-persistence` codec layer), and the storage Request → Response
//! round-trip, correlation, error boundary and lifecycle gates are proven;
//! FSS is not a separate component — `Syss` is the file/storage-system
//! boundary. Phase 7G.1: **storage root + save/load/recovery integration**
//! — [`SySS::initialize`] resolves and establishes the platform storage
//! root (validate, create root + categories, startup crash recovery, fail
//! the boot on error), owns the [`Syss`] handle, and threads the root down
//! the storage boundary; the Storage Service performs the real `Save` /
//! `Load` / `Recover` operations through the existing `Syss`/`Pss`/
//! `Recovery` stack, answered as correlated Responses. Phase 7H: **active
//! objects** — the first genuine application-level objects of the platform
//! (the `hub::active_object` module): real [`Workspace`](crate::hub::Workspace),
//! [`Project`](crate::hub::Project), [`Package`](crate::hub::Package) and
//! [`Document`](crate::hub::Document) runtime objects with stable
//! strongly-typed identities ([`crate::hub::ActiveObjectId`], validated by
//! the platform's single name rule) and the validated `Closed ⇄ Open`
//! lifecycle. The workspace service owns the workspace and the platform's
//! active document; the project/package services own their objects (with
//! the `Workspace → Project → Package` association recorded at open time).
//! The document operations of [`PlatformInterface`] — `open`, `close`,
//! `save`, `save_as`, `import`, `export` — are real: they ride the
//! Request → Response path through the Platform Backbone to the Workspace
//! Authority and its service, which persists through the existing
//! Syss/PSS stack. Phase 7H.1: **workspace/project/package persistence** —
//! the workspace, project and package active objects persist through the
//! same existing stack: their state (identity + the recorded parent
//! association) is serialized with the `.ra` text codec and written
//! atomically by Syss into the existing `workspaces/`, `projects/` and
//! `packages/` categories when the object is opened; reopening restores the
//! state and association, closing and shutdown never delete it, and corrupt
//! or conflicting state is rejected deterministically. Phase 7H.2:
//! **settings/history/log persistence + theme/view foundation** — the
//! Settings, History and Log services become real persisted platform-state
//! domains through the same existing stack (settings as one file per key in
//! `settings/`, history as `history/history`, the smallest real structured
//! platform log as `logs/log`), with lazy loading so corrupt state never
//! fails the boot; [`PlatformInterface::theme`] and
//! [`PlatformInterface::view`] become real settings-backed platform-state
//! operations (deterministic defaults, validated values, persistence;
//! `set_theme`/`get_theme`/`set_view`/`get_view` added additively). Theme
//! and view are application state — no rendering; the Frontend consumes
//! them later. Still missing: the device connector and the RA Core Gateway.
//!
//! # Example
//!
//! ```
//! use ra_syss::{Syss, StorageCategory};
//! use std::path::Path;
//!
//! let syss = Syss::new("storage");
//! assert_eq!(syss.root(), Path::new("storage"));
//! assert_eq!(StorageCategory::Artifact.dir_name(), "artifacts");
//! ```
//!
//! # Platform example
//!
//! ```
//! use ra_syss::{Lifecycle, PlatformState, SySS};
//!
//! let platform = SySS::initialize().unwrap();
//! let state = PlatformState::new();
//! assert_eq!(state.lifecycle(), Lifecycle::Starting);
//! platform.shutdown().unwrap();
//! ```
//!
//! # Platform Interface example
//!
//! ```
//! use ra_syss::PlatformInterface;
//!
//! let platform = PlatformInterface::start().unwrap();
//! platform.shutdown().unwrap();
//! ```

/// Convenience alias for [`Result`](std::result::Result) fixed to [`Error`].
pub type Result<T> = std::result::Result<T, Error>;

/// The error type reported by the SySS platform layer.
///
/// The platform surface reports its own errors and converts errors from the
/// layers it builds on — `ra-persistence` — into this type, so callers
/// handle a single error type. The storage-oriented variants
/// ([`Error::InvalidName`], [`Error::NotFound`], [`Error::AlreadyExists`])
/// are owned here since Phase 7C.1: the storage services ([`Syss`], [`Pss`],
/// [`Recovery`]) report through this type and no longer surface
/// `ra_persistence::Result`.
///
/// The [`PlatformInterface`] operations report through this type: the kernel
/// and backbone failures through [`Error::Kernel`], and the active-object
/// conditions of Phase 7H (duplicate open, close-without-open, missing
/// document, invalid target, persistence failure) through
/// [`Error::ActiveObject`].
#[derive(Debug)]
#[non_exhaustive]
pub enum Error {
    /// An error propagated from the `ra-persistence` layer (codec layer,
    /// artifact model).
    Persistence(ra_persistence::Error),
    /// An underlying host I/O failure.
    Io(std::io::Error),
    /// A storage name failed SySS validation.
    InvalidName(String),
    /// A storage entry was not found in the given category.
    NotFound {
        /// The category that was searched.
        category: StorageCategory,
        /// The requested entry name.
        name: String,
    },
    /// A storage entry already exists where one was not expected.
    AlreadyExists {
        /// The category that was searched.
        category: StorageCategory,
        /// The existing entry name.
        name: String,
    },
    /// The platform kernel failed to initialize or verify readiness.
    ///
    /// Opaque: the kernel internals are never exposed outside `ra-syss`; the
    /// caller only receives a description of the failure.
    Kernel(String),
    /// An active-object operation was rejected (Phase 7H).
    ///
    /// Expresses conditions no storage or kernel error can: e.g. opening an
    /// already-open object, closing an object that is not open, or a
    /// document operation offered while no document is open. The detail stays
    /// opaque and deterministic.
    ActiveObject(String),
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Error::Persistence(err) => write!(f, "persistence error: {err}"),
            Error::Io(err) => write!(f, "io error: {err}"),
            Error::InvalidName(name) => write!(f, "invalid storage name: {name:?}"),
            Error::NotFound { category, name } => {
                write!(f, "no {category} entry named {name:?}")
            }
            Error::AlreadyExists { category, name } => {
                write!(f, "{category} entry {name:?} already exists")
            }
            Error::Kernel(message) => write!(f, "platform kernel: {message}"),
            Error::ActiveObject(message) => write!(f, "active object: {message}"),
        }
    }
}

impl std::error::Error for Error {}

impl From<ra_persistence::Error> for Error {
    fn from(err: ra_persistence::Error) -> Error {
        Error::Persistence(err)
    }
}

impl From<std::io::Error> for Error {
    fn from(err: std::io::Error) -> Error {
        Error::Io(err)
    }
}

impl From<crate::hub::HubError> for Error {
    fn from(err: crate::hub::HubError) -> Error {
        Error::Kernel(err.to_string())
    }
}

/// Returns the current Unix time in milliseconds.
///
/// Test-only helper shared by the module-level unit tests; mirrors the
/// `ra-persistence` implementation.
#[cfg(test)]
pub(crate) fn unix_millis() -> u64 {
    std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map(|d| d.as_millis() as u64)
        .unwrap_or(0)
}

/// Test helpers shared by the module-level unit tests.
#[cfg(test)]
pub(crate) mod test_support {
    use std::fs;
    use std::path::{Path, PathBuf};
    use std::sync::atomic::{AtomicU64, Ordering};

    /// A process-wide counter that makes every temporary directory unique.
    ///
    /// Without it, tests running in parallel in the same process could
    /// collide on a directory name when they start within the same
    /// millisecond, causing one test's cleanup to delete another test's data.
    static NEXT_TEMP_DIR: AtomicU64 = AtomicU64::new(0);

    /// An automatically cleaned-up temporary directory for tests.
    pub(crate) struct TempDir(PathBuf);

    impl TempDir {
        /// Creates a fresh, unique temporary directory.
        pub(crate) fn new(tag: &str) -> TempDir {
            let path = std::env::temp_dir().join(format!(
                "ra-syss-{tag}-{}-{}-{}",
                std::process::id(),
                crate::unix_millis(),
                NEXT_TEMP_DIR.fetch_add(1, Ordering::Relaxed)
            ));
            let _ = fs::remove_dir_all(&path);
            fs::create_dir_all(&path).expect("failed to create test temp dir");
            TempDir(path)
        }

        /// Returns the root path of this temporary directory.
        pub(crate) fn path(&self) -> &Path {
            &self.0
        }
    }

    impl Drop for TempDir {
        fn drop(&mut self) {
            let _ = fs::remove_dir_all(&self.0);
        }
    }
}
